> [!IMPORTANT]
> `team_dev_E_yadoru` は**絶対に編集しないこと**（読み取り専用として扱う）

**ファイル全体を以下に置き換える：**

```java
package com.example.demo.repository;

import java.time.LocalDate;
import java.util.List;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import com.example.demo.entity.Reservation;

public interface ReservationRepository extends JpaRepository<Reservation, Integer> {
	List<Reservation> findByUserIdOrderByReservedAtDesc(Integer userId);

	List<Reservation> findByCheckInDateAfter(LocalDate checkInDate);

	List<Reservation> findByUserFamilyNameContainingAndUserGivenNameContaining(String fn, String gn);

	@Query("SELECT r FROM Reservation r WHERE "
			+ "(:name IS NULL OR :name = '' OR "
			+ " CONCAT(r.user.familyName, r.user.givenName) LIKE CONCAT('%', :name, '%') OR "
			+ " CONCAT(r.user.familyName, ' ', r.user.givenName) LIKE CONCAT('%', :name, '%') OR "
			+ " r.user.familyName LIKE CONCAT('%', :name, '%') OR "
			+ " r.user.givenName LIKE CONCAT('%', :name, '%')) AND "
			+ "(:acmName IS NULL OR :acmName = '' OR r.plan.accommodation.name LIKE CONCAT('%', :acmName, '%')) AND "
			+ "(:checkInDate IS NULL OR r.checkInDate = :checkInDate) "
			+ "ORDER BY r.reservedAt DESC")
	List<Reservation> searchReservations(
			@Param("name") String name,
			@Param("acmName") String acmName,
			@Param("checkInDate") LocalDate checkInDate);

	@Query("SELECT COUNT(r) FROM Reservation r WHERE r.user.id = :userId AND r.plan.accommodation.id = :acmId AND r.isCancelled = false AND r.checkInDate < :checkOut AND r.checkOutDate > :checkIn")
	long countDuplicateReservations(
			@Param("userId") Integer userId,
			@Param("acmId") Integer acmId,
			@Param("checkIn") LocalDate checkIn,
			@Param("checkOut") LocalDate checkOut);
}
```

```java
package com.example.demo.repository;

import java.time.LocalDate;
import java.util.List;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import com.example.demo.dto.AccommodationDto;
import com.example.demo.entity.Accommodation;

public interface AcmRepository extends JpaRepository<Accommodation, Integer> {
	List<Accommodation> findByPrefectureId(Integer prefectureId);

	List<Accommodation> findByNameContaining(String name);

	List<Accommodation> findByPrefectureIdAndNameContaining(Integer prefectureId, String name);

	@Query(" SELECT new com.example.demo.dto.AccommodationDto"
			+ " ("
			+ " ps1.plan.accommodation.id,"
			+ " ps1.plan.accommodation.name,"
			+ " ps1.plan.accommodation.acmCategory.name,"
			+ " CONCAT(ps1.plan.accommodation.prefecture.name , ps1.plan.accommodation.address),"
			+ " MIN(ps1.plan.taxIncludePrice)"
			+ ")"
			+ " FROM PlanStock ps1"
			+ " GROUP BY "
			+ " ps1.plan.accommodation.id,"
			+ " ps1.plan.accommodation.name, "
			+ " ps1.plan.accommodation.acmCategory.name, "
			+ " CONCAT(ps1.plan.accommodation.prefecture.name , ps1.plan.accommodation.address)")
	List<AccommodationDto> findAllDto();

	@Query("SELECT new com.example.demo.dto.AccommodationDto"
			+ "("
			+ "ps1.plan.accommodation.id,"
			+ "ps1.plan.accommodation.name,"
			+ "ps1.plan.accommodation.acmCategory.name,"
			+ "CONCAT(ps1.plan.accommodation.prefecture.name , ps1.plan.accommodation.address),"
			+ "MIN(ps1.plan.taxIncludePrice)"
			+ ") "
			+ "FROM PlanStock ps1 "
			+ "WHERE "
			+ "(SELECT COUNT(*) "
			+ "FROM PlanStock ps2 "
			+ "WHERE "
			+ "ps1.plan.id = ps2.plan.id "
			+ "AND COALESCE(ps2.stock , 0) > 0 "
			+ "AND ps2.targetDate BETWEEN :checkIn AND :checkOut "
			+ ") > :innDays "
			+ "AND (:prefId IS NULL OR ps1.plan.accommodation.prefecture.id = :prefId) "
			+ "AND (:categoryId IS NULL OR ps1.plan.accommodation.acmCategory.id = :categoryId) "
			+ "AND (:name IS NULL OR :name = '' OR ps1.plan.accommodation.name LIKE CONCAT('%', :name , '%')) "
			+ "AND (:peopleCount IS NULL OR ps1.plan.peopleCount >= :peopleCount) "
			+ "GROUP BY "
			+ "ps1.plan.accommodation.id,"
			+ "ps1.plan.accommodation.name,"
			+ "ps1.plan.accommodation.acmCategory.name,"
			+ "CONCAT(ps1.plan.accommodation.prefecture.name , ps1.plan.accommodation.address)")
	List<AccommodationDto> findByAllSearch(
			@Param("name") String name,
			@Param("prefId") Integer prefectureId,
			@Param("categoryId") Integer categoryId,
			@Param("checkIn") LocalDate checkInDate,
			@Param("checkOut") LocalDate checkOutDate,
			@Param("peopleCount") Integer peopleCount,
			@Param("innDays") Integer innDays);

}
```

```java
package com.example.demo.controller;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

import com.example.demo.entity.Reservation;
import com.example.demo.repository.ReservationRepository;
import com.example.demo.repository.UserRepository;
import com.example.demo.service.ReservationService;

@Controller
public class AdminResController {

	private final ReservationRepository reservationRepository;
	private final UserRepository userRepository;
	private final ReservationService reservationService;

	public AdminResController(ReservationRepository reservationRepository, UserRepository userRepository,
			ReservationService reservationService) {
		this.reservationRepository = reservationRepository;
		this.userRepository = userRepository;
		this.reservationService = reservationService;
	}

	// 予約一覧画面の表示
	@GetMapping("/admin/reservations")
	public String listRes(
			@RequestParam(defaultValue = "") String name,
			@RequestParam(defaultValue = "") String acmName,
			@RequestParam(required = false) LocalDate checkInDate,
			Model model) {

		// 予約の絞り込みと検索結果の表示
		List<Reservation> reservationList = reservationRepository.searchReservations(name, acmName, checkInDate);

		model.addAttribute("name", name);
		model.addAttribute("acmName", acmName);
		model.addAttribute("checkInDate", checkInDate);
		model.addAttribute("reservations", reservationList);

		return "admin/reservationList";

	}

	// 予約の詳細画面の表示
	@GetMapping("/reservations/detail/{resId}")
	public String showResDetail(
			@PathVariable Integer resId,
			Model model) {

		Reservation reservation = reservationRepository.findById(resId).get();
		LocalDateTime reserved_at = LocalDateTime.now();
		model.addAttribute("res", reservation);
		model.addAttribute("reserved_at", reserved_at);

		return "admin/reservation_detail";

	}

	// 予約のキャンセル処理
	@PostMapping("/reservations/archive/{resId}")
	public String archiveRes(@PathVariable int resId) {

		reservationService.cancelReservationByAdmin(resId);

		// 予約一覧画面へリダイレクト
		return "redirect:/admin/reservations";
	}
}
```

```java
package com.example.demo.controller;

import java.time.LocalDate;
import java.util.List;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

import com.example.demo.entity.User;
import com.example.demo.repository.UserRepository;

@Controller
public class AdminUserController {

	private final UserRepository userRepository;

	public AdminUserController(
			UserRepository userRepository) {
		this.userRepository = userRepository;

	}

	// 会員一覧画面の表示
	@GetMapping("/admin/users")
	public String show(
			@RequestParam(defaultValue = "") String keyword,
			Model model) {

		// 会員の絞り込みと検索結果の表示
		List<User> userList;
		if (keyword.length() > 0) {
			userList = userRepository.searchByName(keyword);
		} else {
			userList = userRepository.findAll();
		}
		model.addAttribute("keyword", keyword);
		model.addAttribute("users", userList);

		return "admin/user_list";

	}

	// 会員の詳細画面の表示
	@GetMapping("/admin/users/{userId}")
	public String detail(
			@PathVariable Integer userId,
			Model model) {

		User user = userRepository.findById(userId).get();
		model.addAttribute("user", user);
		return "admin/user_detail";
	}

	// 会員の情報をアーカイブ処理
	@PostMapping("/admin/users/{userId}/archive")
	public String archive(@PathVariable int userId) {
		User user = userRepository.findById(userId).get();
		user.setDeactivatedAt(LocalDate.now());
		userRepository.save(user);

		// 会員一覧画面へリダイレクト
		return "redirect:/admin/users";
	}
}
```

```java
package com.example.demo.controller;

import java.time.LocalTime;
import java.util.ArrayList;
import java.util.List;

import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;

import com.example.demo.entity.Accommodation;
import com.example.demo.entity.AcmCategory;
import com.example.demo.entity.Plan;
import com.example.demo.entity.Prefecture;
import com.example.demo.model.AcmList;
import com.example.demo.repository.AcmCategoryRepository;
import com.example.demo.repository.AcmRepository;
import com.example.demo.repository.PlanRepository;
import com.example.demo.repository.PrefectureRepository;

@Controller
public class AdminAcmController {
	private final AcmRepository acmRepository;
	private final PrefectureRepository prefectureRepository;
	private final AcmCategoryRepository acmCategoryRepository;
	private final PlanRepository planRepository;

	public AdminAcmController(AcmRepository acmRepository, PrefectureRepository prefectureRepository,
			AcmCategoryRepository acmCategoryRepository, PlanRepository planRepository) {
		this.acmRepository = acmRepository;
		this.prefectureRepository = prefectureRepository;
		this.acmCategoryRepository = acmCategoryRepository;
		this.planRepository = planRepository;

	}

	// 宿一覧
	@GetMapping("/admin/accommodations")
	public String show(
			@RequestParam(defaultValue = "0") Integer prefectureId,
			@RequestParam(defaultValue = "") String acmName,
			Model model) {

		// 検索セレクトボックス用都道府県リスト
		List<Prefecture> preList = prefectureRepository.findAll();
		model.addAttribute("preList", preList);

		// 宿情報一覧 リスト
		List<AcmList> acmList = new ArrayList<>();

		List<Accommodation> matchedList;
		if (prefectureId > 0 && acmName.length() > 0) {
			matchedList = acmRepository.findByPrefectureIdAndNameContaining(prefectureId, acmName);
		} else if (prefectureId > 0) {
			matchedList = acmRepository.findByPrefectureId(prefectureId);
		} else if (acmName.length() > 0) {
			matchedList = acmRepository.findByNameContaining(acmName);
		} else {
			matchedList = acmRepository.findAll();
		}

		for (Accommodation acm : matchedList) {
			Prefecture pre = null;
			if (acm.getPrefecture() != null) {
				pre = prefectureRepository.findById(acm.getPrefecture().getId()).orElse(null);
			}
			AcmCategory cate = null;
			if (acm.getAcmCategory() != null) {
				cate = acmCategoryRepository.findById(acm.getAcmCategory().getId()).orElse(null);
			}
			List<Plan> plan = planRepository.findByAccommodationIdAndIsArchivedFalse(acm.getId());

			AcmList acmListInfo = new AcmList(acm.getId(), acm.getName(),
					pre != null ? pre.getName() : "",
					cate != null ? cate.getName() : "",
					plan.size());
			acmList.add(acmListInfo);
		}

		model.addAttribute("prefectureId", prefectureId);
		model.addAttribute("acmName", acmName);
		model.addAttribute("acmList", acmList);

		return "admin/acm_list";
	}

	// 宿追加
	@GetMapping("/admin/accommodations/add")
	public String create(Model model) {

		// セレクトボックス用都道府県リスト
		List<Prefecture> preList = prefectureRepository.findAll();
		model.addAttribute("preList", preList);

		// セレクトボックス用宿カテゴリーリスト
		List<AcmCategory> categoryList = acmCategoryRepository.findAll();
		model.addAttribute("categoryList", categoryList);

		return "admin/acm_new";
	}

	@PostMapping("/admin/accommodations/add")
	public String store(
			@RequestParam String name,
			@RequestParam String postalCode,
			@RequestParam Integer prefectureId,
			@RequestParam String address,
			@RequestParam String phoneNumber,
			@RequestParam(required = false) String checkInTime,
			@RequestParam(required = false) String checkOutTime,
			@RequestParam Integer typeOfAcm,
			RedirectAttributes redirectAttributes) {

		List<String> message = new ArrayList<>();
		if (name.length() == 0) {
			message.add("宿名");
		}
		if (postalCode.length() == 0) {
			message.add("郵便番号");
		}
		if (address.length() == 0) {
			message.add("住所(市区町村以降)");
		}
		if (phoneNumber.length() == 0) {
			message.add("電話番号");
		}

		LocalTime parsedCheckInTime = null;
		if (checkInTime == null || checkInTime.isEmpty()) {
			message.add("チェックイン時間");
		} else {
			try {
				parsedCheckInTime = LocalTime.parse(checkInTime);
			} catch (Exception e) {
				message.add("チェックイン時間");
			}
		}

		LocalTime parsedCheckOutTime = null;
		if (checkOutTime == null || checkOutTime.isEmpty()) {
			message.add("チェックアウト時間");
		} else {
			try {
				parsedCheckOutTime = LocalTime.parse(checkOutTime);
			} catch (Exception e) {
				message.add("チェックアウト時間");
			}
		}

		if (message.size() > 0) {
			redirectAttributes.addFlashAttribute("messages", message);

			redirectAttributes.addFlashAttribute("name", name);
			redirectAttributes.addFlashAttribute("postalCode", postalCode);
			redirectAttributes.addFlashAttribute("prefectureId", prefectureId);
			redirectAttributes.addFlashAttribute("address", address);
			redirectAttributes.addFlashAttribute("phoneNumber", phoneNumber);
			redirectAttributes.addFlashAttribute("checkInTime", checkInTime);
			redirectAttributes.addFlashAttribute("checkOutTime", checkOutTime);
			redirectAttributes.addFlashAttribute("typeOfAcm", typeOfAcm);

			return "redirect:/admin/accommodations/add";
		} else {

			Prefecture prefecture = prefectureRepository.findById(prefectureId).get();
			AcmCategory acmCategory = acmCategoryRepository.findById(typeOfAcm).get();
			Accommodation acm = new Accommodation(
					prefecture,
					name,
					postalCode,
					address,
					phoneNumber,
					parsedCheckInTime,
					parsedCheckOutTime,
					acmCategory);
			acmRepository.save(acm);

			return "redirect:/admin/accommodations";
		}
	}

	// 宿詳細
	@GetMapping("/admin/accommodations/detail/{acmId}")
	public String detail(
```

## ステップ10：`PlanService.java`

**ファイルパス：** `src/main/java/com/example/demo/service/PlanService.java`

**ファイル全体を以下に置き換える：**

```java
package com.example.demo.service;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.example.demo.entity.Plan;
import com.example.demo.entity.PlanStock;
import com.example.demo.repository.PlanRepository;
import com.example.demo.repository.PlanStockRepository;

@Service
public class PlanService {

    private PlanRepository planRepository;
    private PlanStockRepository planStockRepository;

    public PlanService(PlanRepository planRepository, PlanStockRepository planStockRepository) {
        this.planRepository = planRepository;
        this.planStockRepository = planStockRepository;
    }

    // 新規プラン登録→プラン在庫の枠を新規作成する
    @Transactional
    public void createNewPlan(Plan plan, Integer initialStock) {
        // プランを保存する
        Plan savedPlan = planRepository.save(plan);

        // 今日を起点に90日分のプラン在庫の枠を生成する
        for (int i = 0; i < 90; i++) {
            PlanStock stock = new PlanStock();
            stock.setPlan(savedPlan);
            stock.setTargetDate(LocalDate.now().plusDays(i));
            stock.setStock(initialStock);
            planStockRepository.save(stock);
        }
    }

    // プランのアーカイブ処理
    public void archivePlan(Integer planId) {
        Plan targetPlan = planRepository.findById(planId).get();

        // アーカイブ済みかを確認
        if (!(targetPlan.getIsArchived())) {
            return;
        }

        List<PlanStock> remPS = planStockRepository.findByPlanIdAndTargetDateAfter(planId, LocalDate.now());

        // 今日以降にプランが登録されているかを確認する
        if (!(remPS.isEmpty())) {
            return;
        }

        targetPlan.setIsArchived(true);
        planRepository.save(targetPlan);

    }

    public List<Plan> getAvailablePlans(Integer accommodationId, LocalDate checkInDate, LocalDate checkOutDate) {

        if (checkInDate == null || checkOutDate == null) {
            return List.of();
        }

        // 1. 必要泊数を計算
        long requiredNights = ChronoUnit.DAYS.between(checkInDate, checkOutDate);
        if (requiredNights <= 0) {
            return List.of(); // 不正な日付範囲の場合は空リストを返す
        }

        // 2. チェックアウト日は在庫不要なため、終了日を調整
        LocalDate endDate = checkOutDate.minusDays(1);

        // 3. 宿泊施設に紐づく全プランの対象期間の在庫データを1回のSQLで一括取得
        List<PlanStock> allStocks = planStockRepository.findByPlanAccommodationIdAndTargetDateBetween(
                accommodationId, checkInDate, endDate);

        // 4. 取得した在庫データをプランごとにグループ化
        Map<Plan, List<PlanStock>> stocksByPlan = allStocks.stream()
                .collect(Collectors.groupingBy(PlanStock::getPlan));

        // 5. 条件を満たすプランのみをフィルタリング
        return stocksByPlan.entrySet().stream()
                .filter(entry -> {
                    List<PlanStock> planStocks = entry.getValue();

                    // 条件①: 滞在するすべての日の在庫レコードが存在するか
                    if (planStocks.size() < requiredNights) {
                        return false;
                    }

                    // 条件②: すべての日のstockが1以上あるか
                    boolean hasAvailableRoomEveryDay = planStocks.stream()
                            .allMatch(stockItem -> stockItem.getStock() != null && stockItem.getStock() > 0);

                    return hasAvailableRoomEveryDay;
                })
                .map(Map.Entry::getKey) // 条件をクリアしたPlanオブジェクトを抽出
                .collect(Collectors.toList());
    }

}
```
